// aquarium-standalone.iff.txt — L4-wrapped form of the aquarium level for wf_game -L.
//
// The aquarium level (docs/plans/2026-09-30-aquarium-level.md): a 55 gal acrylic tank at
// WORLD_SCALE 10, sand, rock, a bubble-tip anemone and a swimming fish.
//
// Build (the standard pipeline emits both inner and standalone):
//   task aquarium-level
//
// Run:
//   task run-aquarium
//
// Pools are snowgoons-class (generous). With the ten followers (82 actors) the room and object memory are raised to 3 MB and 400 KB:
// the one-fish level's 1 MB room heap ran out at 82 actors ("lmalloc ... out of memory, Named Room #0").
{ 'L4'
    { 'ALGN' .align( 2048 ) }
    { 'RAM'
        'OBJD' 400000l
        'PERM' 1000000l
        'ROOM' 3000000l
        'FLAG' 1l 1l                    // doomstick, bungeecam (snowgoons defaults)
    }
    { 'ALGN' .align( 2048 ) }
    [ "../aquarium.iff" ]
}
