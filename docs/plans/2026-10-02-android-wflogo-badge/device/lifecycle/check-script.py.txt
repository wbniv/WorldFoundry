def adb(*args,check=True):
 r=subprocess.run(A+list(args),capture_output=True,check=check,timeout=40);return r.stdout

def pkg(game):return 'org.worldfoundry.wf_game'+('' if game=='snowgoons' else '.'+game)
def pid(p):return adb('shell','pidof',p,check=False).decode().strip()
def start(p):
 out=adb('shell','am','start','-W','-a','android.intent.action.MAIN','-c','android.intent.category.LEANBACK_LAUNCHER','-n',p+'/android.app.NativeActivity').decode()
 assert 'Status: ok' in out,out

def capture(game,stage):
 if stage in ('cold','back-reopen') and (ROOT/f'android/app/src/{game}/assets/layout.json').exists():
  adb('shell','input','keyevent','KEYCODE_BACK');time.sleep(1)
 data=adb('exec-out','screencap','-p');(OUT/f'{game}-{stage}.png').write_bytes(data)
 im=Image.open(io.BytesIO(data)).convert('RGB');colors=im.getcolors(im.width*im.height)

 resumed=adb('shell','dumpsys','activity','activities').decode()
 assert any(pkg(game)+'/android.app.NativeActivity' in line for line in resumed.splitlines() if 'ResumedActivity' in line or 'topResumedActivity' in line),game+' not foreground'
 return len(colors)
